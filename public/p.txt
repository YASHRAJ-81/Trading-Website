  <script>
    let selected = { direction: null, timeframe: null, profit: null };
    const availablePairs = ['BTCUSDT', 'LTCUSDT', 'ETHUSDT', 'BNBUSDT', 'XRPUSDT', 'SOLUSDT', 'DOGEUSDT', 'ADAUSDT'];
    let currentPair = 'BTCUSDT';
    let tradeRunning = false;
    let currentTradeData = null;
    let timerInterval = null;

    // Show hint for available balance below amount input
    const balanceHint = document.getElementById('balanceHint');
    let userBalance = 0;

    // Initialization references for trade details container & timer
    const tradeDetailsContainer = document.getElementById('tradeDetailsContainer');
    const timerEl = document.getElementById('timer');

    // Initialize pair dropdown
    async function renderPairList() {
      const container = document.getElementById('pairDropdown');
      container.innerHTML = '';
      for (const pair of availablePairs) {
        const div = document.createElement('div');
        div.textContent = `${pair.slice(0, 3)}/${pair.slice(3)}`;
        div.className = 'select-none text-white hover:bg-gray-600 rounded px-2 py-1 cursor-pointer';
        div.onclick = async () => {
          toggleDropdown();
          await updatePageForPair(pair, true);
        };
        container.appendChild(div);
      }
    }

    function toggleDropdown() {
      const dropdown = document.getElementById('pairDropdown');
      const btn = document.getElementById('pairDropdownBtn');
      const isShown = dropdown.classList.toggle('show');
      btn.setAttribute('aria-expanded', isShown);
    }

    // Show pair & update chart/info
    async function updatePageForPair(pair, userClick = false) {
      currentPair = pair;
      document.getElementById('pairDropdownBtn').textContent = `${pair.slice(0, 3)}/${pair.slice(3)}`;
      document.getElementById('tvChart').src = `https://www.tradingview.com/widgetembed/?symbol=BINANCE:${pair}&interval=1&theme=dark&hide_top_toolbar=1&hide_side_toolbar=1&locale=en`;
      try {
        const ticker = await fetch(`https://api.binance.com/api/v3/ticker/24hr?symbol=${pair}`).then(r => r.json());
        const klines = await fetch(`https://api.binance.com/api/v3/klines?symbol=${pair}&interval=1d&limit=365`).then(r => r.json());
        const k5 = await fetch(`https://api.binance.com/api/v3/klines?symbol=${pair}&interval=5m&limit=2`).then(r => r.json());

        const open5m = +k5[0][1];
        const close5m = +k5[0][4];
        const high52 = Math.max(...klines.map(d => +d[2]));
        const low52 = Math.min(...klines.map(d => +d[3]));

        document.getElementById('open').textContent = open5m.toFixed(6);
        document.getElementById('close').textContent = close5m.toFixed(6);
        document.getElementById('weekHigh').textContent = high52.toFixed(6);
        document.getElementById('weekLow').textContent = low52.toFixed(6);

        colorize(document.getElementById('open'), open5m, close5m);
        colorize(document.getElementById('close'), close5m, open5m);
        colorize(document.getElementById('weekHigh'), +ticker.lastPrice, high52);
        colorize(document.getElementById('weekLow'), +ticker.lastPrice, low52);
      } catch (err) {
        console.warn('Error fetching coin data', err);
      }
    }

    function colorize(el, current, ref) {
      el.classList.remove('price-up', 'price-down');
      if (current > ref) el.classList.add('price-up');
      else if (current < ref) el.classList.add('price-down');
    }

    // Handle trade direction button with scroll down
    function handleDirectionClick(direction) {
      if (tradeRunning) return;
      selectOption('direction', direction);
      tradeDetailsContainer.classList.remove('hidden');
      tradeDetailsContainer.scrollIntoView({ behavior: 'smooth', block: 'start' });
    }

    // Select timeframe or profit%
    function selectOption(type, value) {
      if (tradeRunning) return;
      selected[type] = value;
      const ids = {
        direction: ['btn-up', 'btn-down'],
        timeframe: ['tf-60', 'tf-90', 'tf-120', 'tf-180'],
        profit: ['p-30', 'p-50', 'p-70', 'p-100'],
      }
      ids[type].forEach(id => {
        const btn = document.getElementById(id);
        if (btn) btn.classList.remove('active-btn', 'bg-yellow-600');
      });
      const btnId = type === 'direction'
        ? (value === 'UP' ? 'btn-up' : 'btn-down')
        : type === 'timeframe'
          ? `tf-${value}`
          : `p-${value}`;
      const btn = document.getElementById(btnId);
      if (btn) btn.classList.add('active-btn', 'bg-yellow-600');
    }

    // Disable or enable inputs during trade
    function setInputsDisabled(disabled) {
      document.getElementById('placeOrderBtn').disabled = disabled;
      document.getElementById('tradeAmount').disabled = disabled;
      document.getElementById('btn-up').disabled = disabled;
      document.getElementById('btn-down').disabled = disabled;
      ['tf-60', 'tf-90', 'tf-120', 'tf-180', 'p-30', 'p-50', 'p-70', 'p-100'].forEach(id => {
        const btn = document.getElementById(id);
        if (btn) btn.disabled = disabled;
      });
    }

    // Submit trade request to backend
    async function submitTrade() {
      if (tradeRunning) {
        alert('Please wait until the current trade finishes.');
        return;
      }

      const amountEl = document.getElementById('tradeAmount');
      const amount = parseFloat(amountEl.value);
      if (!selected.direction || !selected.timeframe || !selected.profit || !amount || isNaN(amount) || amount <= 0) {
        showError('Please fill all fields correctly.');
        return;
      }
      if (amount > userBalance) {
        showError(`Insufficient balance. Your current balance is ${userBalance.toFixed(2)}.`);
        return;
      }
      clearError();

      const tradeData = {
        pair: `${currentPair.slice(0, 3)}/${currentPair.slice(3)}`,
        direction: selected.direction === 'UP' ? 'Buy UP' : 'Buy DOWN',
        amount,
        profitPercent: selected.profit,
        timeframe: selected.timeframe,
      };

      tradeRunning = true;
      setInputsDisabled(true);
      currentTradeData = null;
      startTimer(selected.timeframe);

      try {
        const res = await fetch('/place-trade', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(tradeData)
        });
        const data = await res.json();

        if (!data.success) {
          showError(data.message || 'Trade failed.');
          tradeRunning = false;
          setInputsDisabled(false);
          clearTimer();
          return;
        }

        currentTradeData = data;
        // Do NOT refresh balance or fetch trades here; wait for timer!
      } catch (err) {
        showError(`Error placing trade: ${err.message}`);
        tradeRunning = false;
        setInputsDisabled(false);
        clearTimer();
      }
    }

    // Finalize trade after timer ends
    async function finalizeTrade() {
      if (!currentTradeData) return;

      timerEl.textContent = '⏳ Trade completed.';
      setTimeout(() => timerEl.classList.add('hidden'), 3000);

      await refreshProfileBalance();
      fetchTradeHistory();

      tradeRunning = false;
      setInputsDisabled(false);
      currentTradeData = null;

      // --- RESET FIELDS after trade ---
      selected.direction = null;
      selected.timeframe = null;
      selected.profit = null;
      ['btn-up', 'btn-down', 'tf-60', 'tf-90', 'tf-120', 'tf-180', 'p-30', 'p-50', 'p-70', 'p-100'].forEach(id => {
        const btn = document.getElementById(id);
        if (btn) btn.classList.remove('active-btn', 'bg-yellow-600');
      });
      document.getElementById('tradeAmount').value = '';
      resetTimerUI();
      // Optionally, scroll to top of trade panel
      // tradeDetailsContainer.scrollIntoView({behavior: 'smooth', block: 'start'});
      // --- END RESET ---
    }

    // Show error message
    function showError(msg) {
      const el = document.getElementById('errorMessage');
      el.textContent = msg;
      el.classList.remove('hidden');
    }
    function clearError() {
      const el = document.getElementById('errorMessage');
      el.textContent = '';
      el.classList.add('hidden');
    }

    // Timer logic with countdown and completion triggering finalizeTrade()
    function startTimer(seconds) {
      timerEl.classList.remove('hidden');
      let remaining = seconds;

      function formatTime(s) {
        const m = Math.floor(s / 60).toString().padStart(2, '0');
        const ss = (s % 60).toString().padStart(2, '0');
        return `${m}:${ss}`;
      }

      timerEl.textContent = `⏱ ${formatTime(remaining)} remaining`;

      clearTimer();
      timerInterval = setInterval(() => {
        remaining--;
        if (remaining < 0) {
          clearInterval(timerInterval);
          finalizeTrade();
        } else {
          timerEl.textContent = `⏱ ${formatTime(remaining)} remaining`;
        }
      }, 1000);
    }

    function clearTimer() {
      if (timerInterval) {
        clearInterval(timerInterval);
        timerInterval = null;
      }
    }
    function resetTimerUI() {
      timerEl.classList.add('hidden');
      timerEl.textContent = '';
    }

    function updateBalanceHint() {
      balanceHint.textContent = `Your current balance: ${userBalance.toFixed(2)}`;
    }

    // Refresh user balance from server/profile
    async function refreshProfileBalance() {
      try {
        const res = await fetch('/profile-data');
        if (!res.ok) throw new Error('Unauthorized');
        const data = await res.json();
        userBalance = parseFloat(data.balance) || 0;
        updateBalanceHint();
      } catch (err) {
        console.error('Failed to refresh profile:', err);
      }
    }

    // Manage trade history UI
    const tradeHistoryEl = document.getElementById('tradeHistory');
    const detailsEl = document.getElementById('expandedTradeDetails');

    // Load full trade history from server but render only 4 recent trades
    async function fetchTradeHistory() {
      try {
        const res = await fetch('/trade-history');
        if (!res.ok) throw new Error('Unauthorized');
        const data = await res.json();
        const trades = data.trades || [];
        renderTrades(trades.slice(0, 4)); // Show only latest 4 trades here
      } catch (err) {
        tradeHistoryEl.innerHTML = `<div class="text-gray-400 text-center">Failed to load trades.</div>`;
        console.warn(err);
      }
    }

    function renderTrades(trades) {
      tradeHistoryEl.innerHTML = '';
      detailsEl.classList.add('hidden');
      detailsEl.textContent = '';

      if (!trades.length) {
        tradeHistoryEl.innerHTML = '<div class="text-gray-400 text-center">No trades yet.</div>';
        return;
      }

      trades.forEach(trade => {
        const div = document.createElement('div');
        div.className = 'trade-entry';
        div.title = 'Click to toggle trade details';
        div.innerHTML = `
          <div>
            ${trade.pair} &nbsp;
            <span>${trade.direction === 'Buy UP' ? '↑' : '↓'}</span> &nbsp;
            <span>${trade.timeframe}s</span>
          </div>
          <div class="${trade.result === 'WIN' ? 'text-green-400' : 'text-red-400'}">
            ${trade.result === 'WIN' ? '+' : '-'}${Math.abs(parseFloat(trade.profit)).toFixed(2)}
            <span class="trade-arrow">▼</span>
          </div>
        `;
        div.onclick = () => toggleDetails(trade);
        tradeHistoryEl.appendChild(div);
      });
    }

    function toggleDetails(trade) {
      if (detailsEl.dataset.tradeId === String(trade.id)) {
        detailsEl.classList.add('hidden');
        detailsEl.textContent = '';
        delete detailsEl.dataset.tradeId;
        return;
      }
      detailsEl.dataset.tradeId = trade.id;
      detailsEl.classList.remove('hidden');
      detailsEl.textContent = `
Pair: ${trade.pair}
Direction: ${trade.direction}
Amount: ${trade.amount}
Timeframe: ${trade.timeframe}s
Profit %: ${trade.profitPercent}%
Result: ${trade.result}
Profit/Loss: ${trade.profit}
Timestamp: ${new Date(trade.timestamp).toLocaleString()}
      `.trim();
    }

    function toggleInputs(disabled) {
      setInputsDisabled(disabled);
    }

    function goToDashboard() {
      window.location.href = '../dashboard.html';
    }

    async function init() {
      await refreshProfileBalance();
      await renderPairList();
      await updatePageForPair(currentPair);
      await fetchTradeHistory();
    }
    init();
  </script>

const express = require('express');
const bodyParser = require('body-parser');
const session = require('express-session');
const svgCaptcha = require('svg-captcha');
const bcrypt = require('bcryptjs');
const low = require('lowdb');
const FileSync = require('lowdb/adapters/FileSync');
const path = require('path');

const app = express();
const port = 3000;

// Database setup
const adapter = new FileSync('db.json');
const db = low(adapter);
db.defaults({ users: [], trades: [] }).write();

app.use(bodyParser.urlencoded({ extended: true }));
app.use(bodyParser.json());
app.use(express.static('public'));
app.use(session({ secret: 'auth-secret', resave: false, saveUninitialized: true }));

// Homepage
app.get('/', (req, res) => {
  res.sendFile(path.join(__dirname, 'public/index.html'));
});

// Captcha generation
app.get('/captcha', (req, res) => {
  const captcha = svgCaptcha.create({
    noise: 2,
    color: true,
    background: '#ffffff',
    width: 150,
    height: 50,
    fontSize: 45,
    charPreset: 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789',
  });

  let svg = captcha.data;
  svg = svg.replace(/(<text[^>]*?)fill="[^"]*"/g, '$1fill="black"');
  svg = svg.replace(/(<text[^>]*?)style="[^"]*fill:[^;"]*;?[^"]*"/g, '$1style="fill:black"');
  svg = svg.replace(/(<text[^>]*?)y="\d+"/g, '$1y="40"');

  req.session.captcha = captcha.text;
  res.type('svg').send(svg);
});

// Register
app.post('/register', (req, res) => {
  const { email, password, confirmPassword } = req.body;
  const inviteCode = 'nikki23';

  if (password !== confirmPassword) {
    return res.json({ success: false, message: 'Passwords do not match.' });
  }

  const existing = db.get('users').find({ email }).value();
  if (existing) {
    return res.json({ success: false, message: 'User already exists.' });
  }

  const hashed = bcrypt.hashSync(password, 10);
  db.get('users').push({
    email,
    password: hashed,
    inviteCode,
    balance: 2000,
    trades: []
  }).write();

  return res.json({ success: true });
});

// Login
app.post('/login', (req, res) => {
  const { email, password, captcha } = req.body;

  if (captcha !== req.session.captcha) {
    return res.json({ success: false, message: 'Invalid captcha.' });
  }

  const user = db.get('users').find({ email }).value();
  if (!user || !bcrypt.compareSync(password, user.password)) {
    return res.json({ success: false, message: 'Invalid email or password.' });
  }

  req.session.user = {
    email,
    inviteCode: user.inviteCode || 'nikki23'
  };

  return res.json({ success: true });
});

// Profile data
app.get('/profile-data', (req, res) => {
  if (!req.session.user) return res.status(401).json({ error: 'Not logged in' });

  const user = db.get('users').find({ email: req.session.user.email }).value();
  if (!user) return res.status(404).json({ error: 'User not found' });

  const { email, inviteCode, bank, balance } = user;
  res.json({ email, inviteCode, bank, balance });
});

// Reset password
app.post('/reset-password', (req, res) => {
  const { email, captcha, newPassword } = req.body;

  if (captcha !== req.session.captcha) {
    return res.json({ success: false, message: 'Invalid captcha.' });
  }

  const user = db.get('users').find({ email }).value();
  if (!user) return res.json({ success: false, message: 'Email not found.' });

  const hashed = bcrypt.hashSync(newPassword, 10);
  db.get('users').find({ email }).assign({ password: hashed }).write();

  return res.json({ success: true });
});

// Withdraw password
app.post('/update-withdraw-password', (req, res) => {
  const { withdrawPassword } = req.body;

  if (!req.session.user) return res.status(401).json({ error: "Unauthorized" });
  if (!withdrawPassword || withdrawPassword.length < 4)
    return res.status(400).json({ error: "Invalid password" });

  db.get('users')
    .find({ email: req.session.user.email })
    .assign({ withdrawPassword })
    .write();

  res.json({ success: true });
});

// Bind bank card
app.post('/bind-bank-card', (req, res) => {
  const { bankName, accountNumber, accountHolderName } = req.body;
  if (!req.session.user) return res.status(401).json({ message: 'Not logged in' });

  if (!bankName || !accountNumber || !accountHolderName) {
    return res.status(400).json({ message: 'Missing bank card information' });
  }

  db.get('users')
    .find({ email: req.session.user.email })
    .assign({
      bank: {
        bankName,
        accountNumber,
        accountHolderName
      }
    })
    .write();

  return res.json({ success: true });
});

// Updated trade route with full validation and logic
app.post('/place-trade', (req, res) => {
  if (!req.session.user) return res.status(401).json({ success: false, message: 'Not logged in' });

  const { pair, direction, amount, profitPercent, timeframe } = req.body;

  if (
    !pair || !direction || !amount || profitPercent === undefined || !timeframe
  ) {
    return res.json({ success: false, message: 'Missing trade parameters' });
  }

  const user = db.get('users').find({ email: req.session.user.email }).value();
  if (!user) return res.status(404).json({ success: false, message: 'User not found' });

  const tradeAmount = parseFloat(amount);
  const profitPct = parseFloat(profitPercent);
  const tradeTimeframe = parseInt(timeframe, 10);

  if (
    isNaN(tradeAmount) || tradeAmount <= 0 ||
    tradeAmount > user.balance
  ) {
    return res.json({ success: false, message: 'Invalid trade amount or insufficient balance' });
  }

  if (isNaN(profitPct) || profitPct < 0) {
    return res.json({ success: false, message: 'Invalid profit percent' });
  }

  if (isNaN(tradeTimeframe) || tradeTimeframe <= 0) {
    return res.json({ success: false, message: 'Invalid timeframe' });
  }

  // Change here: Always treat trade as a win (profit) regardless of pair or direction
  let isWin = true;

  // Calculate profit or loss (always profit now)
  const profitAmount = tradeAmount * (profitPct / 100);

  // Update balance after trade result
  const newBalance = user.balance + profitAmount;

  // Create trade object
  const trade = {
    id: Date.now(),
    pair,
    direction,
    amount: tradeAmount,
    profit: profitAmount.toFixed(2),
    result: 'WIN',
    profitPercent: profitPct,
    timeframe: tradeTimeframe,
    timestamp: new Date().toISOString()
  };

  // Update user balance and add trade
  db.get('users')
    .find({ email: req.session.user.email })
    .assign({ balance: newBalance })
    .get('trades')
    .push(trade)
    .write();

  return res.json({
    success: true,
    trade,
    balance: newBalance.toFixed(2),
  });
});

// Change password
app.post('/change-password', (req, res) => {
  const { oldPassword, newPassword, confirmPassword } = req.body;

  if (!req.session.user) return res.status(401).json({ success: false, message: 'Not logged in' });

  const user = db.get('users').find({ email: req.session.user.email }).value();
  if (!user) return res.status(404).json({ success: false, message: 'User not found' });

  if (!bcrypt.compareSync(oldPassword, user.password)) {
    return res.json({ success: false, message: 'Old password is incorrect' });
  }

  if (newPassword !== confirmPassword) {
    return res.json({ success: false, message: 'New passwords do not match' });
  }

  const hashed = bcrypt.hashSync(newPassword, 10);
  db.get('users').find({ email: user.email }).assign({ password: hashed }).write();

  return res.json({ success: true });
});
